Database SQL Setup (One-Time)

Paste this SQL in your Supabase SQL Editor to create the quotations, company_settings, and item_catalog tables.

-- 1. Create quotations table
CREATE TABLE IF NOT EXISTS public.quotations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  quote_no TEXT,
  quote_date DATE NOT NULL DEFAULT CURRENT_DATE,
  company_name TEXT,
  company_logo_text TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  client_name TEXT,
  client_contact TEXT,
  client_phone TEXT,
  project_name TEXT,
  site_location TEXT,
  items JSONB DEFAULT '[]'::jsonb,
  gst_rate NUMERIC DEFAULT 18,
  discount NUMERIC DEFAULT 0,
  validity INTEGER DEFAULT 15,
  payment_terms TEXT,
  notes TEXT,
  subtotal NUMERIC DEFAULT 0,
  total NUMERIC DEFAULT 0
);

-- 2. Create company_settings table
CREATE TABLE IF NOT EXISTS public.company_settings (
  id TEXT PRIMARY KEY DEFAULT 'default',
  company_name TEXT,
  company_logo_text TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Create reusable item_catalog table
CREATE TABLE IF NOT EXISTS public.item_catalog (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  description TEXT NOT NULL UNIQUE,
  rate NUMERIC NOT NULL DEFAULT 0,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Enable Row Level Security & grant access
ALTER TABLE public.quotations ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.company_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.item_catalog ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public access to quotations" ON public.quotations FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to company_settings" ON public.company_settings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to item_catalog" ON public.item_catalog FOR ALL USING (true) WITH CHECK (true);
