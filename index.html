Database SQL Setup (One-Time)

Paste this SQL in your Supabase SQL Editor to create the quotations, company_settings, and item_catalog tables.

-- 1. Create quotations table
CREATE TABLE IF NOT EXISTS public.quotations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  quote_no TEXT,
  quote_date DATE NOT NULL DEFAULT CURRENT_DATE,
  company_name TEXT,
  logo_mode TEXT DEFAULT 'ai',
  ai_logo_style TEXT DEFAULT 'industrial',
  ai_logo_color TEXT DEFAULT 'amber_slate',
  ai_logo_symbol TEXT DEFAULT 'auto',
  ai_logo_seed NUMERIC DEFAULT 1,
  ai_custom_svg TEXT,
  company_logo_text TEXT,
  company_logo_img TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  client_name TEXT,
  client_contact TEXT,
  client_phone TEXT,
  project_name TEXT,
  site_location TEXT,
  items JSONB DEFAULT '[]'::jsonb,
  gst_mode TEXT DEFAULT '18',
  gst_rate NUMERIC DEFAULT 18,
  discount NUMERIC DEFAULT 0,
  validity INTEGER DEFAULT 15,
  payment_terms TEXT,
  notes TEXT,
  subtotal NUMERIC DEFAULT 0,
  total NUMERIC DEFAULT 0
);

-- 2. Create company_settings table
CREATE TABLE IF NOT EXISTS public.company_settings (
  id TEXT PRIMARY KEY DEFAULT 'default',
  company_name TEXT,
  logo_mode TEXT DEFAULT 'ai',
  ai_logo_style TEXT DEFAULT 'industrial',
  ai_logo_color TEXT DEFAULT 'amber_slate',
  ai_logo_symbol TEXT DEFAULT 'auto',
  ai_logo_seed NUMERIC DEFAULT 1,
  ai_custom_svg TEXT,
  company_logo_text TEXT,
  company_logo_img TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Create reusable item_catalog table
CREATE TABLE IF NOT EXISTS public.item_catalog (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  description TEXT NOT NULL UNIQUE,
  rate NUMERIC NOT NULL DEFAULT 0,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Ensure extra columns exist on existing databases (safe migrations)
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS logo_mode TEXT DEFAULT 'ai';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS ai_logo_style TEXT DEFAULT 'industrial';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS ai_logo_color TEXT DEFAULT 'amber_slate';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS ai_logo_symbol TEXT DEFAULT 'auto';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS ai_logo_seed NUMERIC DEFAULT 1;
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS ai_custom_svg TEXT;
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS company_logo_img TEXT;
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS gst_mode TEXT DEFAULT '18';

ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS logo_mode TEXT DEFAULT 'ai';
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS ai_logo_style TEXT DEFAULT 'industrial';
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS ai_logo_color TEXT DEFAULT 'amber_slate';
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS ai_logo_symbol TEXT DEFAULT 'auto';
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS ai_logo_seed NUMERIC DEFAULT 1;
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS ai_custom_svg TEXT;
ALTER TABLE public.company_settings ADD COLUMN IF NOT EXISTS company_logo_img TEXT;

-- 5. Enable Row Level Security & grant access
ALTER TABLE public.quotations ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.company_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.item_catalog ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public access to quotations" ON public.quotations FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to company_settings" ON public.company_settings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to item_catalog" ON public.item_catalog FOR ALL USING (true) WITH CHECK (true);
