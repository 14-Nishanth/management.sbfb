Database SQL Setup (One-Time)

Paste this SQL in your Supabase SQL Editor to create the quotations, company_settings, item_catalog, gst_registry, and custom_layouts tables.

-- 1. Create quotations table
CREATE TABLE IF NOT EXISTS public.quotations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  quote_no TEXT,
  quote_date DATE NOT NULL DEFAULT CURRENT_DATE,
  company_name TEXT,
  logo_mode TEXT DEFAULT 'ai',
  ai_logo_style TEXT DEFAULT 'industrial',
  ai_logo_color TEXT DEFAULT 'amber_slate',
  ai_logo_symbol TEXT DEFAULT 'auto',
  ai_logo_seed NUMERIC DEFAULT 1,
  ai_custom_svg TEXT,
  company_logo_text TEXT,
  company_logo_img TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  client_name TEXT,
  client_gst TEXT,
  client_contact TEXT,
  client_phone TEXT,
  project_name TEXT,
  site_location TEXT,
  items JSONB DEFAULT '[]'::jsonb,
  gst_mode TEXT DEFAULT '18',
  gst_rate NUMERIC DEFAULT 18,
  discount NUMERIC DEFAULT 0,
  validity INTEGER DEFAULT 15,
  payment_terms TEXT,
  notes TEXT,
  subtotal NUMERIC DEFAULT 0,
  total NUMERIC DEFAULT 0,
  template_theme TEXT DEFAULT 'modern',
  custom_layout_config JSONB
);

-- 2. Create company_settings table
CREATE TABLE IF NOT EXISTS public.company_settings (
  id TEXT PRIMARY KEY DEFAULT 'default',
  company_name TEXT,
  logo_mode TEXT DEFAULT 'ai',
  ai_logo_style TEXT DEFAULT 'industrial',
  ai_logo_color TEXT DEFAULT 'amber_slate',
  ai_logo_symbol TEXT DEFAULT 'auto',
  ai_logo_seed NUMERIC DEFAULT 1,
  ai_custom_svg TEXT,
  company_logo_text TEXT,
  company_logo_img TEXT,
  company_phone TEXT,
  company_email TEXT,
  company_gst TEXT,
  company_address TEXT,
  bank_name TEXT,
  bank_account TEXT,
  bank_ifsc TEXT,
  bank_branch TEXT,
  bank_upi TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Create reusable item_catalog table
CREATE TABLE IF NOT EXISTS public.item_catalog (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  description TEXT NOT NULL UNIQUE,
  rate NUMERIC NOT NULL DEFAULT 0,
  unit TEXT NOT NULL DEFAULT 'pcs',
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Create master gst_registry table
CREATE TABLE IF NOT EXISTS public.gst_registry (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  gstin TEXT NOT NULL UNIQUE,
  legal_name TEXT NOT NULL,
  trade_name TEXT,
  state_code TEXT,
  state_name TEXT,
  contact TEXT,
  address TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 5. Create custom_layouts table (User Bill Layouts Saved to Cloud)
CREATE TABLE IF NOT EXISTS public.custom_layouts (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  category TEXT DEFAULT 'modern',
  config JSONB NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 6. Ensure extra columns exist on existing databases (safe migrations)
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS client_gst TEXT;
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS logo_mode TEXT DEFAULT 'ai';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS template_theme TEXT DEFAULT 'modern';
ALTER TABLE public.quotations ADD COLUMN IF NOT EXISTS custom_layout_config JSONB;
ALTER TABLE public.item_catalog ADD COLUMN IF NOT EXISTS unit TEXT DEFAULT 'pcs';

-- 7. Enable Row Level Security & grant access
ALTER TABLE public.quotations ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.company_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.item_catalog ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.gst_registry ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.custom_layouts ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public access to quotations" ON public.quotations FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to company_settings" ON public.company_settings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to item_catalog" ON public.item_catalog FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to gst_registry" ON public.gst_registry FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access to custom_layouts" ON public.custom_layouts FOR ALL USING (true) WITH CHECK (true);
